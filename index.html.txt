        // Web Audio Context setup for synth sounds (safari-proofed)
        let audioCtx = null;
        function getAudioContext() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
            return audioCtx;
        }

        // Sound effect generator
        function playTone(freq, type, duration, delay = 0) {
            if (isMuted) return;
            try {
                const ctx = getAudioContext();
                if (ctx.state === 'suspended') {
                    ctx.resume();
                }
                setTimeout(() => {
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.type = type;
                    osc.frequency.setValueAtTime(freq, ctx.currentTime);
                    
                    gain.gain.setValueAtTime(0.15, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + duration);
                    
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    osc.start();
                    osc.stop(ctx.currentTime + duration);
                }, delay * 1000);
            } catch (e) {
                console.log("Audio play error", e);
            }
        }

        function playCorrectSound() {

        // START GAME
        function startGame() {
            score = 0;
            completedLevels = {};
            scorePerLevel = {};
            document.getElementById('top-score').innerText = `⭐ النقاط: ${score}`;
            
            // Switch Screen
            document.getElementById('welcome-screen').classList.add('hidden');
            document.getElementById('map-screen').classList.remove('hidden');
            
            getAudioContext(); // initializes web audio context safely
            playVictorySound();
            renderMap();
        }

        // RENDER THE INTERACTIVE MAP
        function renderMap() {
            const grid = document.getElementById('stages-grid');
            grid.innerHTML = '';
            
            let totalUnlocked = 0;
            
            levels.forEach((lvl, idx) => {
                const isUnlocked = idx === 0 || completedLevels[idx - 1];
                const isCompleted = completedLevels[idx];
                
                if (isUnlocked) totalUnlocked++;
                
                const card = document.createElement('div');
                card.className = `p-5 rounded-2xl border text-center transition-all duration-300 relative overflow-hidden ${
                    isCompleted 
                    ? 'bg-emerald-900/80 border-amber-500 shadow-md shadow-emerald-950/50 cursor-pointer hover:scale-105' 
                    : isUnlocked 
                        ? 'bg-emerald-950/90 border-emerald-600 shadow cursor-pointer hover:border-amber-500 hover:scale-105' 
                        : 'bg-emerald-950/30 border-emerald-900/60 opacity-55 cursor-not-allowed select-none'
                }`;
                
                if (isUnlocked) {
                    card.onclick = () => loadLevel(idx);
                }
                
                // Card Inner HTML
                card.innerHTML = `
                    <div class="text-3xl mb-2">${isCompleted ? '✅' : isUnlocked ? '🔓' : '🔒'}</div>
                    <span class="text-[10px] text-amber-400 block font-bold mb-1">${lvl.badge}</span>
                    <h5 class="text-sm font-bold text-slate-100 line-clamp-2">${lvl.title}</h5>
                    ${isCompleted ? `<div class="text-[10px] text-emerald-400 font-bold mt-2">مكتملة (+${scorePerLevel[idx]})</div>` : ''}
                `;
                
                grid.appendChild(card);
            });

            // Update Progress Bar
            const progressPercent = (Object.keys(completedLevels).length / levels.length) * 100;
            document.getElementById('overall-progress').style.width = `${progressPercent}%`;

            // If all levels are completed, show certificate trigger
            if (Object.keys(completedLevels).length === levels.length) {
                document.getElementById('finish-game-area').style.display = 'block';
            } else {
                document.getElementById('finish-game-area').style.display = 'none';
            }
        }

        function exitToMap() {
            document.getElementById('gameplay-screen').classList.add('hidden');
            document.getElementById('map-screen').classList.remove('hidden');
            renderMap();
        }

        // LOAD LEVEL DATA
        let activeLevelScenarios = [];
        let activeScenarioIndex = 0;
        let activeLevelScore = 0;
        let activeMatchingPairs = [];
        let activeMatchingSelected = null;
        let activeSortingItems = [];
        let activeSortingIndex = 0;

        function loadLevel(idx) {
            currentLevelIndex = idx;
            const lvl = levels[idx];
            
            document.getElementById('map-screen').classList.add('hidden');
            document.getElementById('gameplay-screen').classList.remove('hidden');
            
            document.getElementById('level-number-badge').innerText = lvl.badge;
            document.getElementById('level-title').innerText = lvl.title;
            
            // Set up Study View
            document.getElementById('study-card').classList.remove('hidden');
            document.getElementById('challenge-card').classList.add('hidden');
            document.getElementById('lesson-text').innerHTML = lvl.lesson;
        }

        // TRANSITION FROM LESSON TO CHALLENGE
        function startLevelChallenge() {
            document.getElementById('study-card').classList.add('hidden');
            document.getElementById('challenge-card').classList.remove('hidden');
            
            // Reset Challenge View State
            document.getElementById('feedback-box').classList.add('hidden');
            
            const lvl = levels[currentLevelIndex];
            activeLevelScore = 0;
            
            // Build Workspace based on challenge type
            if (lvl.type === "scenario") {
                activeLevelScenarios = JSON.parse(JSON.stringify(lvl.challenge.scenarios));
                activeScenarioIndex = 0;
                renderScenarioChallenge();
            } else if (lvl.type === "matching") {
                activeMatchingPairs = JSON.parse(JSON.stringify(lvl.challenge.pairs));
                activeMatchingSelected = null;
                renderMatchingChallenge();
            } else if (lvl.type === "sorting-belt") {
                activeSortingItems = JSON.parse(JSON.stringify(lvl.challenge.items));
                activeSortingIndex = 0;
                renderSortingChallenge();
            }
        }

        // TYPE A: SCENARIO CHALLENGE RENDER
        function renderScenarioChallenge() {
            const workspace = document.getElementById('interactive-workspace');
            const scenario = activeLevelScenarios[activeScenarioIndex];
            const actions = document.getElementById('level-actions');
            
            document.getElementById('feedback-box').classList.add('hidden');
            
            workspace.innerHTML = `
                <div class="space-y-4">
                    <p class="text-xs text-amber-400 font-bold uppercase tracking-widest mb-1">الموقف الفقهي (${activeScenarioIndex + 1} من ${activeLevelScenarios.length})</p>
                    <blockquote class="text-base md:text-lg font-bold border-r-4 border-amber-500 pr-4 py-2 bg-emerald-950/40 rounded-l-xl text-slate-100">
                        "${scenario.text}"
                    </blockquote>
                    <p class="text-xs text-slate-400 mt-2">ما التوجيه والقرار الفقهي المناسب والموافق للكتيب العلمي؟</p>
                    
                    <div class="grid grid-cols-1 gap-3 mt-4" id="options-container">
                        ${scenario.options.map((opt, i) => `
                            <button onclick="submitScenarioAnswer(${i})" class="w-full text-right p-4 rounded-xl border border-emerald-700/60 bg-emerald-900/30 hover:bg-emerald-900/80 active:scale-98 transition-all hover:border-amber-500 text-sm md:text-base font-semibold">
                                🔹 ${opt.text}
                            </button>
                        `).join('')}
                    </div>
                </div>
            `;
            
            actions.innerHTML = ''; // Next scenario buttons are driven by feedback step
        }

        function submitScenarioAnswer(optIndex) {
            const scenario = activeLevelScenarios[activeScenarioIndex];
            const isCorrect = scenario.options[optIndex].correct;
            const buttons = document.getElementById('options-container').getElementsByTagName('button');
            
            // Disable all buttons and color code them
            for (let i = 0; i < buttons.length; i++) {
                buttons[i].disabled = true;
                if (scenario.options[i].correct) {
                    buttons[i].className = "w-full text-right p-4 rounded-xl border-2 border-emerald-500 bg-emerald-950 text-emerald-300 text-sm md:text-base font-bold";
                } else {
                    buttons[i].className = "w-full text-right p-4 rounded-xl border border-red-500/50 bg-red-950/20 text-red-300 opacity-50 text-sm md:text-base";
                }
            }

            // Show feedback
            const feedbackBox = document.getElementById('feedback-box');
            const fbIcon = document.getElementById('feedback-icon');
            const fbTitle = document.getElementById('feedback-title');
            const fbDesc = document.getElementById('feedback-desc');
            
            feedbackBox.classList.remove('hidden');
            if (isCorrect) {
                playCorrectSound();
                activeLevelScore += 10;
                feedbackBox.className = "border border-emerald-500 bg-emerald-950/60 p-4 rounded-2xl transition-all duration-300";
                fbIcon.innerText = "⭐";
                fbTitle.className = "font-bold text-emerald-400";
                fbTitle.innerText = "إجابة موفقة!";
                fbDesc.innerText = scenario.feedback;
            } else {
                playIncorrectSound();
                feedbackBox.className = "border border-red-500/50 bg-red-950/40 p-4 rounded-2xl transition-all duration-300";
                fbIcon.innerText = "❌";
                fbTitle.className = "font-bold text-red-400";
                fbTitle.innerText = "تحتاج لمراجعة المسألة!";
                fbDesc.innerText = scenario.feedback + " تذكّر الالتزام الحرفي بنصوص الكتاب الفقهي.";

            const actions = document.getElementById('level-actions');
            const isLast = activeScenarioIndex === activeLevelScenarios.length - 1;
            
            actions.innerHTML = `
                <button onclick="${isLast ? 'finishLevel()' : 'nextScenario()'}" class="bg-amber-500 hover:bg-amber-600 text-emerald-950 font-bold px-6 py-2.5 rounded-xl shadow-lg transition-all active:scale-95 text-sm flex items-center gap-2">
                    <span>${isLast ? 'تأكيد وحفظ النتيجة' : 'الموقف الفقهي التالي'}</span>
                    <span>➡️</span>

        function nextScenario() {
            activeScenarioIndex++;

        // TYPE B: MATCHING CHALLENGE RENDER (Strict Single-Tap Matching for smooth mobile experience)
        let matchingSelectedState = null; // {type: 'item'|'answer', index: number}
        let matchedIndices = {}; // index of item -> true

        function renderMatchingChallenge() {
            const workspace = document.getElementById('interactive-workspace');
            const actions = document.getElementById('level-actions');
            
            matchingSelectedState = null;
            matchedIndices = {};
            
            // Shuffle answers for randomized matching
            const shuffledAnswers = activeMatchingPairs.map((pair, idx) => ({ text: pair.answer, originalIdx: idx }));
            shuffledAnswers.sort(() => Math.random() - 0.5);

            workspace.innerHTML = `
                <div class="space-y-4">
                    <p class="text-xs text-amber-400 font-bold mb-1">الربط الفقهي: اختر حالة من العمود الأيمن ثم اختر ما يطابقها في العمود الأيسر:</p>
                    
                    <div class="grid grid-cols-2 gap-4">
                        <!-- Left Column (Items) -->
                        <div class="space-y-2.5" id="match-items-col">
                            ${activeMatchingPairs.map((pair, idx) => `
                                <button id="match-item-${idx}" onclick="selectMatchingElement('item', ${idx})" class="w-full text-right p-3 rounded-xl border border-emerald-700 bg-emerald-900/20 text-xs md:text-sm transition-all focus:outline-none">
                                    📌 ${pair.item}
                                </button>
                            `).join('')}
                        </div>
                        
                        <!-- Right Column (Answers) -->
                        <div class="space-y-2.5" id="match-answers-col">
                            ${shuffledAnswers.map((ans, sIdx) => `
                                <button id="match-ans-${ans.originalIdx}" onclick="selectMatchingElement('answer', ${ans.originalIdx})" class="w-full text-right p-3 rounded-xl border border-emerald-700 bg-emerald-900/20 text-xs md:text-sm transition-all focus:outline-none">
                                    💡 ${ans.text}
                                </button>
                            `).join('')}
                        </div>
                    </div>
                </div>
            `;
            
            actions.innerHTML = `
                <button id="matching-submit-btn" disabled class="bg-emerald-800 text-emerald-500 opacity-50 font-bold px-6 py-2.5 rounded-xl cursor-not-allowed text-sm">
                    اربط كافة المسائل للمتابعة
                </button>
            `;
        }

        function selectMatchingElement(type, originalIdx) {
            // If already matched, ignore
            if (matchedIndices[originalIdx]) return;

            // Clear previous highlighting on non-matched components
            resetHighlighting();

            if (!matchingSelectedState) {
                // First selection
                matchingSelectedState = { type, originalIdx };
                highlightElement(type, originalIdx, 'border-amber-500 bg-amber-500/10 text-amber-300');
            } else {
                // Second selection
                if (matchingSelectedState.type === type) {
                    // Changed mind, select new one of same column
                    matchingSelectedState = { type, originalIdx };
                    highlightElement(type, originalIdx, 'border-amber-500 bg-amber-500/10 text-amber-300');
                } else {
                    // Matching Attempt!
                    if (matchingSelectedState.originalIdx === originalIdx) {
                        // CORRECT Match!
                        playCorrectSound();
                        activeLevelScore += 10;
                        matchedIndices[originalIdx] = true;
                        
                        highlightElement('item', originalIdx, 'border-emerald-500 bg-emerald-950 text-emerald-400 font-bold opacity-60 pointer-events-none');
                        highlightElement('answer', originalIdx, 'border-emerald-500 bg-emerald-950 text-emerald-400 font-bold opacity-60 pointer-events-none');
                        
                        // Disable buttons
                        document.getElementById(`match-item-${originalIdx}`).disabled = true;
                        document.getElementById(`match-ans-${originalIdx}`).disabled = true;
                    } else {
                        // INCORRECT Match
                        playIncorrectSound();
                        highlightElement(type, originalIdx, 'border-red-500 bg-red-950 text-red-300');
                        highlightElement(matchingSelectedState.type, matchingSelectedState.originalIdx, 'border-red-500 bg-red-950 text-red-300');
                        
                        setTimeout(() => {
                            resetHighlighting();
                        }, 1000);
                    }
                    matchingSelectedState = null;
                }
            }

            // Check if all matched
            if (Object.keys(matchedIndices).length === activeMatchingPairs.length) {
                const actions = document.getElementById('level-actions');
                actions.innerHTML = `
                    <button onclick="finishLevel()" class="bg-amber-500 hover:bg-amber-600 text-emerald-950 font-bold px-6 py-2.5 rounded-xl shadow-lg transition-all active:scale-95 text-sm">
                        إنهاء المرحلة وحفظ النتيجة ✨
                    </button>
                `;
            }
        }

        function highlightElement(type, idx, classes) {
            const elId = type === 'item' ? `match-item-${idx}` : `match-ans-${idx}`;
            const el = document.getElementById(elId);
            if (el) {
                el.className = `w-full text-right p-3 rounded-xl border ${classes} text-xs md:text-sm transition-all focus:outline-none`;
            }
        }

        function resetHighlighting() {
            activeMatchingPairs.forEach((_, idx) => {
                if (!matchedIndices[idx]) {
                    highlightElement('item', idx, 'border-emerald-700 bg-emerald-900/20 text-slate-100');
                    highlightElement('answer', idx, 'border-emerald-700 bg-emerald-900/20 text-slate-100');
                }
            });
        }


        // TYPE C: CONVEYOR BELT / FAST SORTING CHALLENGE

                        <span class="text-4xl block mb-2">💧</span>
                        <span class="text-lg font-bold text-slate-100" id="sorting-item-name">${currentItem.name}</span>

                    <!-- Dual Option Buttons (Perfect for Mobile Tapping) -->

                            🚨 يُفطّر الصائم
                        </button>
                        <button onclick="submitSortingAnswer(false)" class="bg-emerald-900/40 hover:bg-emerald-800/60 border border-emerald-500/40 text-emerald-300 font-bold py-3.5 rounded-xl active:scale-95 transition-all text-sm">
                            ✅ لا يُفطّر (صحيح)
                        </button>
                    </div>
                </div>

            actions.innerHTML = '';
        }

        function submitSortingAnswer(claimedIsBreaker) {
            const currentItem = activeSortingItems[activeSortingIndex];
            const isCorrect = currentItem.isBreaker === claimedIsBreaker;
            
            // Provide Sound / Score feedback
            if (isCorrect) {
                playCorrectSound();
                activeLevelScore += 5; // 5 points per sort item
            } else {
                playIncorrectSound();
            }

            // Visual flash/feedback on workspace
            const workspace = document.getElementById('interactive-workspace');
            workspace.classList.add(isCorrect ? 'border-emerald-500' : 'border-red-500');
            
            // Advance or finish
            setTimeout(() => {
                workspace.classList.remove('border-emerald-500', 'border-red-500');
                
                if (activeSortingIndex < activeSortingItems.length - 1) {
                    activeSortingIndex++;
                    renderSortingChallenge();
                } else {
                    finishLevel();
                }
            }, 350);
        }

        // SAVE AND FINISH LEVEL
        function finishLevel() {
            completedLevels[currentLevelIndex] = true;
            scorePerLevel[currentLevelIndex] = activeLevelScore;
            
            // Re-calculate Total Points
            score = Object.values(scorePerLevel).reduce((a, b) => a + b, 0);
            document.getElementById('top-score').innerText = `⭐ النقاط: ${score}`;
            
            exitToMap();
            playVictorySound();
        }

            } else if (score >= 70) {
                rank = "طالب علم نبيه 📚";
            }

        }
